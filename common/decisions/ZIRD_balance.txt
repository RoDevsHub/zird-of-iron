    ZIRD_timeout_till_danzig_mission = {

        allowed = {
            original_tag = GER
        }

        activation = {
            date > 1939.01.01
            AND = {
                NOT = { has_war_with = POL }
                country_exists = POL
            }
        }
        available = {
            has_war_with = POL
        }
        cancel_trigger = {
            OR = {
                has_war_with = POL
                NOT = { country_exists = POL }
            }
        }
        icon = mission_icon     # For GFX_decision_mission_icon

        days_mission_timeout = 366
        selectable_mission = no
        timeout_effect = {
            declare_war_on = {
                target = POL
                type = annex_everything
            }
        }
	}
